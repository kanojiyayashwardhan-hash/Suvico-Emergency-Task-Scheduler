cmake_minimum_required(VERSION 3.20)
project(SuvicoEmergencyTaskScheduler LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(httplib GIT_REPOSITORY https://github.com/yhirose/cpp-httplib.git GIT_TAG v0.18.3)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.11.3)
FetchContent_MakeAvailable(httplib json)
add_executable(suvico src/main.cpp)
target_link_libraries(suvico PRIVATE httplib::httplib nlohmann_json::nlohmann_json)
target_compile_definitions(suvico PRIVATE SUVICO_PUBLIC_DIR="${CMAKE_CURRENT_SOURCE_DIR}/public" SUVICO_DATA_FILE="${CMAKE_CURRENT_SOURCE_DIR}/data/tasks.json")
if(WIN32)
  target_link_libraries(suvico PRIVATE ws2_32)
endif()
